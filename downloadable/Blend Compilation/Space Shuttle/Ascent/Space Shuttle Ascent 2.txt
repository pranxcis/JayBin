import bpy, math

# ═════════════════════════════════════════════════════════════
# SCENE RESET (KEEP SHUTTLE LOGIC SAFE)
# ═════════════════════════════════════════════════════════════

for obj in bpy.data.objects:
    if obj.name != "Shuttle_Root" and not obj.parent:
        obj.hide_set(True)
        obj.hide_render = True

sc = bpy.context.scene
FPS = 24
DURATION = 8
TOTAL_FRAMES = FPS * DURATION

sc.frame_start = 1
sc.frame_end = TOTAL_FRAMES
sc.render.fps = FPS

# ═════════════════════════════════════════════════════════════
# WORLD — ATMOSPHERIC SKY (NOT SPACE)
# ═════════════════════════════════════════════════════════════

world = bpy.data.worlds.new("SkyScene2")
world.use_nodes = True
sc.world = world

nodes = world.node_tree.nodes
links = world.node_tree.links
nodes.clear()

out = nodes.new("ShaderNodeOutputWorld")
bg = nodes.new("ShaderNodeBackground")
sky = nodes.new("ShaderNodeTexSky")
mapping = nodes.new("ShaderNodeMapping")
coord = nodes.new("ShaderNodeTexCoord")

# SKY MODEL (REAL ATMOSPHERE LOOK)
sky.sky_type = 'PREETHAM'
sky.sun_elevation = math.radians(18)   # higher = more sky, less space feel
sky.sun_rotation = math.radians(40)

# Warm morning atmospheric tint (IMPORTANT FIX)
bg.inputs["Color"].default_value = (0.35, 0.55, 0.85, 1)  # blue sky base
bg.inputs["Strength"].default_value = 1.0

# Link sky texture properly (so it actually drives atmosphere)
links.new(coord.outputs["Generated"], mapping.inputs["Vector"])
links.new(mapping.outputs["Vector"], sky.inputs["Vector"])
links.new(sky.outputs["Color"], bg.inputs["Color"])
links.new(bg.outputs["Background"], out.inputs["Surface"])

# ═════════════════════════════════════════════════════════════
# LIGHTING — ATMOSPHERIC DAYLIGHT (NO SPACE FEEL)
# ═════════════════════════════════════════════════════════════

bpy.ops.object.light_add(type='SUN', location=(0, 0, 100))
sun = bpy.context.active_object
sun.name = "Sun_Key"

sun.data.energy = 3.2
sun.data.color = (1.0, 0.92, 0.75)  # warm morning sun

# Lower angle = atmosphere scattering = SKY FEEL
sun.rotation_euler = (
    math.radians(25),   # higher sun than space lighting
    0,
    math.radians(-30)
)

# Soft sky fill (IMPORTANT for atmosphere)
bpy.ops.object.light_add(type='AREA', location=(-40, -20, 60))
fill = bpy.context.active_object
fill.name = "Sky_Fill"

fill.data.energy = 80
fill.data.size = 60
fill.data.color = (0.6, 0.75, 1.0)

# ═════════════════════════════════════════════════════════════
# ASSUME SHUTTLE EXISTS FROM SCENE 1
# ═════════════════════════════════════════════════════════════

shuttle = bpy.data.objects.get("Shuttle_Root")
if not shuttle:
    raise Exception("Shuttle_Root not found. Run Scene 1 first.")

# ═════════════════════════════════════════════════════════════
# SHUTTLE ASCENT — TILTED GRAVITY TURN
# ═════════════════════════════════════════════════════════════

shuttle.location = (0, 0, 0)

# Initial tilt (launch trajectory)
shuttle.rotation_euler = (
    -math.radians(38),  # 30–45 degree ascent
    0,
    math.radians(5)
)
shuttle.keyframe_insert("rotation_euler", frame=1)

# Mid ascent (slight correction)
shuttle.rotation_euler = ( 
    -math.radians(22),
    0,
    math.radians(2)
)
shuttle.keyframe_insert("rotation_euler", frame=240)

# High ascent (nearly vertical)
shuttle.rotation_euler = (
    -math.radians(10),
    0,
    0
)
shuttle.keyframe_insert("rotation_euler", frame=360)

# Motion path (sky climb)
shuttle.location = (0, 0, 0)
shuttle.keyframe_insert("location", frame=1)

shuttle.location = (0, 0, 120)
shuttle.keyframe_insert("location", frame=200)

shuttle.location = (0, 0, 420)
shuttle.keyframe_insert("location", frame=360)

shuttle.location = (0, 0, 900)
shuttle.keyframe_insert("location", frame=460)

# Smooth interpolation
if shuttle.animation_data and shuttle.animation_data.action:
    for fc in shuttle.animation_data.action.fcurves:     
        for kp in fc.keyframe_points:
            kp.interpolation = 'BEZIER'
            kp.easing = 'EASE_IN'

# ═════════════════════════════════════════════════════════════
# CAMERA — CHASE RIG (FOLLOW SHUTTLE)
# ═════════════════════════════════════════════════════════════

bpy.ops.object.camera_add(location=(0, -25, 8))
cam = bpy.context.active_object
cam.name = "Scene2_Camera"

cam.data.lens = 15

# Parent to shuttle for motion match
cam.parent = shuttle
cam.location = (0, -22, 6)

# Track shuttle (always centered)
track = cam.constraints.new(type='DAMPED_TRACK')
track.target = shuttle
track.track_axis = 'TRACK_NEGATIVE_Z'

# Slight DOF for cinematic depth
cam.data.dof.use_dof = True
cam.data.dof.focus_object = shuttle
cam.data.dof.aperture_fstop = 4.5

# ═════════════════════════════════════════════════════════════
# OPTIONAL: CAMERA SHAKE (LAUNCH VIBRATION FEEL)
# ═════════════════════════════════════════════════════════════

cam.keyframe_insert("location", frame=1)
cam.location.x += 0.05
cam.location.y += 0.03
cam.keyframe_insert("location", frame=12)

cam.location.x -= 0.02
cam.location.y -= 0.01
cam.keyframe_insert("location", frame=24)

# ═════════════════════════════════════════════════════════════
# PLUME BOOST (MAKE IT MORE VISIBLE IN SKY)
# ═════════════════════════════════════════════════════════════

plume = bpy.data.objects.get("EnginePlume")
if plume and plume.material_slots:
    mat = plume.material_slots[0].material
    if mat and mat.node_tree:
        for n in mat.node_tree.nodes:
            if n.type == 'BSDF_PRINCIPLED':
                if "Emission Strength" in n.inputs:
                    n.inputs["Emission Strength"].default_value *= 1.4

# Engine glow boost
glow = bpy.data.objects.get("EngineGlow_Light")
if glow:
    glow.data.energy *= 1.5

# ═════════════════════════════════════════════════════════════
# FINAL SETUP
# ═════════════════════════════════════════════════════════════

sc.frame_set(1)

print("=== SCENE 2 READY: SKY ASCENT FOLLOW SHOT ===")
print("Camera: chase + tracked")
print("Shuttle: gravity turn (30–45° → vertical)")
print("Environment: sky-only cinematic ascent")
