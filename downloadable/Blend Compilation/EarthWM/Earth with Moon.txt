import bpy
import math
import mathutils

# ==== SETTINGS ====
EARTH_TEX = r"C:\Users\Francis Balbin\Downloads\Francis\Solar System Blender\Planets Resources\Earth.jpg"
MOON_TEX  = r"C:\Users\Francis Balbin\Downloads\Francis\Solar System Blender\Planets Resources\Moon.jpg"

# ==== CLEAN SCENE ====
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete()
bpy.ops.outliner.orphans_purge(do_recursive=True)

# ==== HELPER: apply texture to object ====
def apply_texture(obj, tex_path):
    mat = bpy.data.materials.new(name=obj.name + "Mat")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    nodes.clear()
    tex = nodes.new('ShaderNodeTexImage')
    tex.image = bpy.data.images.load(tex_path)
    bsdf = nodes.new('ShaderNodeBsdfPrincipled')
    bsdf.inputs['Roughness'].default_value = 0.8
    out = nodes.new('ShaderNodeOutputMaterial')
    mat.node_tree.links.new(tex.outputs['Color'], bsdf.inputs['Base Color'])
    mat.node_tree.links.new(bsdf.outputs['BSDF'], out.inputs['Surface'])
    obj.data.materials.append(mat)

# ==== EARTH ====
bpy.ops.mesh.primitive_uv_sphere_add(segments=64, ring_count=32, radius=1, location=(0, 0, 0))
earth = bpy.context.active_object
earth.name = "Earth"
bpy.ops.object.shade_smooth()
apply_texture(earth, EARTH_TEX)

# Axial tilt + spin animation
earth.rotation_mode = 'XYZ'
earth.rotation_euler = (math.radians(23.5), 0, 0)
earth.keyframe_insert(data_path="rotation_euler", frame=1)
earth.rotation_euler[2] = math.radians(360 * 3)   # 3 full spins
earth.keyframe_insert(data_path="rotation_euler", frame=240)

# ==== MOON (orbits via empty pivot) ====
bpy.ops.object.empty_add(location=(0, 0, 0))
pivot = bpy.context.active_object
pivot.name = "MoonPivot"
pivot.keyframe_insert(data_path="rotation_euler", frame=1)
pivot.rotation_euler[2] = math.radians(360)        # 1 full orbit
pivot.keyframe_insert(data_path="rotation_euler", frame=240)

bpy.ops.mesh.primitive_uv_sphere_add(segments=64, ring_count=32, radius=0.27, location=(3.0, 0, 0))
moon = bpy.context.active_object
moon.name = "Moon"
bpy.ops.object.shade_smooth()
apply_texture(moon, MOON_TEX)
moon.parent = pivot

# ==== BLACK BACKGROUND ====
bpy.context.scene.world.node_tree.nodes["Background"].inputs[0].default_value = (0, 0, 0, 1)
bpy.context.scene.world.node_tree.nodes["Background"].inputs[1].default_value = 0

# ==== SUN LIGHT ====
bpy.ops.object.light_add(type='SUN', location=(10, -10, 8))
bpy.context.active_object.data.energy = 6
bpy.context.active_object.rotation_euler = (math.radians(50), 0, math.radians(45))

# ==== CAMERA ====
bpy.ops.object.camera_add(location=(5, -9, 4))
cam = bpy.context.active_object
cam.name = "Camera"

# Point camera at Earth (origin)
direction = mathutils.Vector((0, 0, 0)) - mathutils.Vector((5, -9, 4))
cam.rotation_euler = direction.to_track_quat('-Z', 'Y').to_euler()

bpy.context.scene.camera = cam
cam.data.lens = 50

# ==== LINEAR INTERPOLATION on both ====
for obj in [earth, pivot]:
    for fc in obj.animation_data.action.fcurves:   
        for kp in fc.keyframe_points:
            kp.interpolation = 'LINEAR'

# ==== RENDER (Cycles, GPU) ====
scene = bpy.context.scene
scene.frame_start, scene.frame_end, scene.render.fps = 1, 240, 24
scene.render.engine = 'CYCLES'
scene.cycles.device = 'GPU'
scene.cycles.samples = 128
scene.cycles.use_adaptive_sampling = True
scene.cycles.use_denoising = True
scene.cycles.denoiser = 'OPENIMAGEDENOISE'
scene.render.use_persistent_data = True  # faster animation rendering
