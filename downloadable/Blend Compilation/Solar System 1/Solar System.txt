# ── HELPERS ───────────────────────────────────────────────────────────────────
def tex(name):
    p = os.path.join(TEXTURE_DIR, name)
    return bpy.data.images.load(p) if os.path.exists(p) else None

def material(name, tex_file, roughness=1.0, emit=0.0, fallback=(0.5, 0.5, 0.5, 1)):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    N, L = mat.node_tree.nodes, mat.node_tree.links
    N.clear()
    out  = N.new('ShaderNodeOutputMaterial'); out.location  = (400, 0)
    bsdf = N.new('ShaderNodeBsdfPrincipled'); bsdf.location = (100, 0)
    bsdf.inputs['Roughness'].default_value = roughness
    bsdf.inputs['Specular IOR Level'].default_value = 0.0 
    
    img = tex(tex_file)
    if img:
        t = N.new('ShaderNodeTexImage'); t.location = (-250, 0); t.image = img
        L.new(t.outputs['Color'], bsdf.inputs['Base Color'])
    else:
        bsdf.inputs['Base Color'].default_value = fallback
    if emit:
        e = N.new('ShaderNodeEmission'); e.location = (100, -200)
        e.inputs['Strength'].default_value = emit
        if img: L.new(t.outputs['Color'], e.inputs['Color'])
        else:   e.inputs['Color'].default_value = fallback
        mx = N.new('ShaderNodeMixShader'); mx.location = (280, 0)
        mx.inputs['Fac'].default_value = 1.0  
        L.new(bsdf.outputs['BSDF'],  mx.inputs[1])
        L.new(e.outputs['Emission'], mx.inputs[2])
        L.new(mx.outputs['Shader'],  out.inputs['Surface'])
    else:
        L.new(bsdf.outputs['BSDF'], out.inputs['Surface'])
    return mat

def sphere(name, r, loc=(0, 0, 0)):
    bpy.ops.mesh.primitive_uv_sphere_add(segments=128, ring_count=64, radius=r, location=loc)
    obj = bpy.context.active_object
    obj.name = name
    bpy.ops.object.shade_smooth()
    return obj

def get_fcurves(action):
    if hasattr(action, 'fcurves'):
        return action.fcurves
    try:
        strip = action.layers[0].strips[0]
        return strip.channelbag(strip.channelbags[0].binding).fcurves
    except (IndexError, AttributeError):
        return []

def set_linear_keys(obj):
    if obj.animation_data and obj.animation_data.action:
        for fc in get_fcurves(obj.animation_data.action):
            for kp in fc.keyframe_points:
                kp.interpolation = 'LINEAR'

def animate_spin(obj, period_frames, total, direction=1):
    period_frames = int(period_frames)
    step = max(1, period_frames // 4)
    for f in range(0, total + period_frames, step):
        obj.rotation_euler[2] = direction * (f / period_frames) * 2 * math.pi
        obj.keyframe_insert(data_path='rotation_euler', frame=f)
    set_linear_keys(obj)

def orbit_guide(r, name):
    bpy.ops.mesh.primitive_torus_add(
        major_radius=r, minor_radius=r * 0.003,
        major_segments=256, minor_segments=8, location=(0, 0, 0))
    t = bpy.context.active_object; t.name = name + "_Orbit"
    mat = bpy.data.materials.new(name + "_Orbit_Mat")
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs['Base Color'].default_value = (0.15, 0.15, 0.2, 1)
    bsdf.inputs['Alpha'].default_value      = 0.3
    mat.blend_method = 'BLEND'
    t.data.materials.append(mat)

# Starting angle for each pivot so the planet faces the camera (positive-X)
# exactly when the camera arrives at its close-up keyframe.
PLANET_START_ANGLE = {
    "Mercury": -4.6822,   # close-up f=310,  period=416
    "Venus":   -3.2427,   # close-up f=545,  period=1056
    "Earth":   -2.8362,   # close-up f=780,  period=1728
    "Mars":    -1.9683,   # close-up f=1015, period=3240
    "Jupiter": -0.3879,   # close-up f=1280, period=20736
    "Saturn":  -0.1873,   # close-up f=1545, period=51840
    "Uranus":  -0.0759,   # close-up f=1790, period=148176

    "Pluto":   -0.0292,   # close-up f=2075, period=446880

saturn_planet = None

for name, r, orbit_r, period, tilt, spin, tex_file, color in PLANETS:
    bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0, 0, 0))
    pivot = bpy.context.active_object; pivot.name = name + "_Pivot"
    period = int(period)
    step   = max(1, period // 8)
    start  = PLANET_START_ANGLE[name]
    for f in range(0, TOTAL_FRAMES + period, step):
        pivot.rotation_euler[2] = start + (f / period) * 2 * math.pi
        pivot.keyframe_insert(data_path='rotation_euler', frame=f)
    set_linear_keys(pivot)

    planet = sphere(name, r, loc=(orbit_r, 0, 0))
    planet.rotation_euler[0] = math.radians(tilt)
    animate_spin(planet, abs(spin), TOTAL_FRAMES, direction=-1 if spin < 0 else 1)
    planet.data.materials.append(material(name + "_Mat", tex_file, fallback=color))
    planet.parent = pivot

    # Orbit guide — visible in viewport but hidden from render
    orbit_guide(orbit_r, name)
    guide = bpy.data.objects[name + "_Orbit"]
    guide.hide_render = True

    # Make 100% invisible in render
    guide.visible_camera = False
    guide.visible_diffuse = False
    guide.visible_glossy = False
    guide.visible_transmission = False
    guide.visible_volume_scatter = False
    guide.visible_shadow = False

    if name == "Saturn":
        saturn_planet = planet
